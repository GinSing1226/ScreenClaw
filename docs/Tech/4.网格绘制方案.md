> 索引：[技术文档目录](./0.技术文档目录.md)

## 需求

- 在截图上绘制百分比坐标网格
- 网格参数可配置（密度、透明度、颜色）
- 坐标数字可配置（密度、小数位、大小、颜色）

---

## 技术方案

使用 **Pillow (PIL)** 进行图片绘制。

### 依赖

```txt
Pillow>=10.0.0
```

## 绘制流程

```
原始截图 (PIL.Image)
    │
    ▼
┌─────────────────────────────────────┐
│  1. 创建半透明图层                   │
│     Image.new("RGBA", size, (0,0,0,0)) │
└──────────────┬──────────────────────┘
               ▼
┌─────────────────────────────────────┐
│  2. 绘制网格线                       │
│     ImageDraw.line()                 │
│     颜色 + 透明度                    │
└──────────────┬──────────────────────┘
               ▼
┌─────────────────────────────────────┐
│  3. 绘制坐标数字                     │
│     ImageDraw.text()                 │
│     在交叉点上方                     │
└──────────────┬──────────────────────┘
               ▼
┌─────────────────────────────────────┐
│  4. 合并图层                         │
│     Image.alpha_composite()          │
└──────────────┬──────────────────────┘
               ▼
带网格的截图
```

## 代码实现

### core/grid.py

```python
from PIL import Image, ImageDraw, ImageFont
from typing import Tuple

class GridRenderer:
    """网格绘制器"""

    def __init__(self, config: dict):
        self.density = config.get("density", 5.0)          # 网格密度（百分比）
        self.grid_opacity = config.get("opacity", 50)      # 网格透明度
        self.grid_color = config.get("color", "#00FF00")   # 网格颜色

        self.number_density = config.get("number_density", 2)    # 数字密度
        self.number_decimal = config.get("number_decimal", 0)    # 小数位
        self.number_size = config.get("number_size", 8)          # 字体大小
        self.number_color = config.get("number_color", "#00FF00")
        self.number_opacity = config.get("number_opacity", 100)

    def draw_grid(self, image: Image.Image) -> Image.Image:
        """
        在图片上绘制网格

        Args:
            image: 原始截图

        Returns:
            带网格的图片
        """
        width, height = image.size

        # 确保图片是RGBA模式
        if image.mode != "RGBA":
            image = image.convert("RGBA")

        # 创建透明图层
        overlay = Image.new("RGBA", image.size, (0, 0, 0, 0))
        draw = ImageDraw.Draw(overlay)

        # 解析颜色
        grid_rgba = self._hex_to_rgba(self.grid_color, self.grid_opacity)
        number_rgba = self._hex_to_rgba(self.number_color, self.number_opacity)

        # 加载字体
        font = self._load_font(self.number_size)

        # 计算网格间距（像素）
        grid_step_x = int(width * self.density / 100)
        grid_step_y = int(height * self.density / 100)

        # 绘制网格线
        self._draw_grid_lines(draw, width, height, grid_step_x, grid_step_y, grid_rgba)

        # 绘制坐标数字
        self._draw_coordinates(draw, width, height, grid_step_x, grid_step_y, font, number_rgba)

        # 合并图层
        result = Image.alpha_composite(image, overlay)

        return result

    def _draw_grid_lines(self, draw: ImageDraw, width: int, height: int,
                         step_x: int, step_y: int, color: Tuple[int, int, int, int]):
        """绘制网格线"""

        # 垂直线
        x = 0
        while x <= width:
            draw.line([(x, 0), (x, height)], fill=color, width=1)
            x += step_x

        # 水平线
        y = 0
        while y <= height:
            draw.line([(0, y), (width, y)], fill=color, width=1)
            y += step_y

    def _draw_coordinates(self, draw: ImageDraw, width: int, height: int,
                          step_x: int, step_y: int, font, color: Tuple[int, int, int, int]):
        """绘制坐标数字"""

        text_step_x = step_x * self.number_density
        text_step_y = step_y * self.number_density

        x = 0
        percent_x = 0
        while x <= width:
            y = 0
            percent_y = 0
            while y <= height:
                # 格式化坐标文本
                text = self._format_coordinate(percent_x, percent_y)

                # 计算文本位置（交叉点上方）
                bbox = draw.textbbox((0, 0), text, font=font)
                text_width = bbox[2] - bbox[0]
                text_height = bbox[3] - bbox[1]

                pos_x = x - text_width // 2
                pos_y = y - text_height - 2  # 上方偏移

                # 绘制文本
                draw.text((pos_x, pos_y), text, fill=color, font=font)

                y += text_step_y
                percent_y += self.density * self.number_density

            x += text_step_x
            percent_x += self.density * self.number_density

    def _format_coordinate(self, x_percent: float, y_percent: float) -> str:
        """格式化坐标文本"""
        if self.number_decimal == 0:
            return f"{int(x_percent)},{int(y_percent)}"
        else:
            format_str = f"{{:.{self.number_decimal}f}}"
            return f"{format_str.format(x_percent)},{format_str.format(y_percent)}"

    def _hex_to_rgba(self, hex_color: str, opacity: int) -> Tuple[int, int, int, int]:
        """HEX颜色转RGBA"""
        hex_color = hex_color.lstrip("#")
        r = int(hex_color[0:2], 16)
        g = int(hex_color[2:4], 16)
        b = int(hex_color[4:6], 16)
        a = int(255 * opacity / 100)
        return (r, g, b, a)

    def _load_font(self, size: int) -> ImageFont:
        """加载系统默认字体"""
        try:
            # Windows 默认字体
            return ImageFont.truetype("arial.ttf", size)
        except:
            # 回退到默认字体
            return ImageFont.load_default()
```

## 使用示例

```python
from PIL import Image

# 加载截图
screenshot = Image.open("screenshot.png")

# 配置网格参数
config = {
    "density": 5.0,
    "opacity": 50,
    "color": "#00FF00",
    "number_density": 2,
    "number_decimal": 0,
    "number_size": 8,
    "number_color": "#00FF00",
    "number_opacity": 100
}

# 绘制网格
renderer = GridRenderer(config)
result = renderer.draw_grid(screenshot)

# 保存
result.save("screenshot_with_grid.png")
```

## 图片压缩

### utils/image.py

```python
from PIL import Image
import io
import base64

def compress_image(image: Image.Image, quality: int = 85,
                   max_width: int = 1920) -> Image.Image:
    """压缩图片"""
    # 缩放
    if image.width > max_width:
        ratio = max_width / image.width
        new_height = int(image.height * ratio)
        image = image.resize((max_width, new_height), Image.LANCZOS)

    return image

def image_to_base64(image: Image.Image, format: str = "PNG") -> str:
    """图片转base64"""
    buffer = io.BytesIO()
    image.save(buffer, format=format)
    return base64.b64encode(buffer.getvalue()).decode("utf-8")

def save_image(image: Image.Image, path: str, quality: int = 85):
    """保存图片"""
    if path.lower().endswith(".jpg") or path.lower().endswith(".jpeg"):
        image.save(path, "JPEG", quality=quality)
    else:
        image.save(path, "PNG")
```
