## 需求

- 向目标窗口注入点击、滑动、输入等操作
- 优先支持后台操作（不抢夺物理键鼠）
- 后台失败时，弹窗确认后使用前台模拟

---

## 方案对比

| 方案 | 原理 | 后台支持 | 兼容性 | 用户体验 |
|------|------|----------|--------|----------|
| **PostMessage** | 发送消息到窗口队列 | ✅ | ⭐⭐⭐ | ⭐⭐⭐⭐⭐ |
| **SendInput** | 模拟硬件输入 | ❌ | ⭐⭐⭐⭐⭐ | ⭐⭐ |

## 方案1：PostMessage

### 原理

向目标窗口的消息队列发送消息，窗口自行处理。

### 支持的操作

| 操作 | 消息序列 |
|------|----------|
| 点击 | WM_LBUTTONDOWN → WM_LBUTTONUP |
| 右键 | WM_RBUTTONDOWN → WM_RBUTTONUP |
| 长按 | WM_LBUTTONDOWN → 等待 → WM_LBUTTONUP |
| 滑动 | WM_LBUTTONDOWN → WM_MOUSEMOVE → WM_LBUTTONUP |
| 按键 | WM_KEYDOWN → WM_KEYUP |

### 代码示例

```python
import win32gui
import win32con
import win32api

class PostMessageInjector:

    def click(self, hwnd: int, x: int, y: int) -> bool:
        """点击操作"""
        lParam = win32api.MAKELONG(x, y)

        # 发送点击消息
        win32gui.PostMessage(hwnd, win32con.WM_LBUTTONDOWN, win32con.MK_LBUTTON, lParam)
        win32gui.PostMessage(hwnd, win32con.WM_LBUTTONUP, 0, lParam)

        return True

    def right_click(self, hwnd: int, x: int, y: int) -> bool:
        """右键点击"""
        lParam = win32api.MAKELONG(x, y)

        win32gui.PostMessage(hwnd, win32con.WM_RBUTTONDOWN, win32con.MK_RBUTTON, lParam)
        win32gui.PostMessage(hwnd, win32con.WM_RBUTTONUP, 0, lParam)

        return True

    def long_press(self, hwnd: int, x: int, y: int, duration_ms: int) -> bool:
        """长按操作"""
        import time
        lParam = win32api.MAKELONG(x, y)

        win32gui.PostMessage(hwnd, win32con.WM_LBUTTONDOWN, win32con.MK_LBUTTON, lParam)
        time.sleep(duration_ms / 1000)
        win32gui.PostMessage(hwnd, win32con.WM_LBUTTONUP, 0, lParam)

        return True

    def swipe(self, hwnd: int, start_x: int, start_y: int, end_x: int, end_y: int) -> bool:
        """滑动操作"""
        lParam_start = win32api.MAKELONG(start_x, start_y)
        lParam_end = win32api.MAKELONG(end_x, end_y)

        win32gui.PostMessage(hwnd, win32con.WM_LBUTTONDOWN, win32con.MK_LBUTTON, lParam_start)
        win32gui.PostMessage(hwnd, win32con.WM_MOUSEMOVE, win32con.MK_LBUTTON, lParam_end)
        win32gui.PostMessage(hwnd, win32con.WM_LBUTTONUP, 0, lParam_end)

        return True

    def key_press(self, hwnd: int, key: str) -> bool:
        """按键操作"""
        vk_code = self._key_to_vk(key)
        if vk_code is None:
            return False

        win32gui.PostMessage(hwnd, win32con.WM_KEYDOWN, vk_code, 0)
        win32gui.PostMessage(hwnd, win32con.WM_KEYUP, vk_code, 0)

        return True
```

### 失败检测

PostMessage 不返回执行结果，如何判断是否成功？

**方案**：截图对比
1. 操作前截图
2. 执行PostMessage
3. 等待一小段时间
4. 操作后截图
5. 对比是否有变化

```python
def detect_postmessage_success(hwnd: int, action_func) -> bool:
    """检测PostMessage是否生效"""
    before = capture_window(hwnd)
    action_func()
    time.sleep(0.5)
    after = capture_window(hwnd)

    # 图片对比
    diff = compare_images(before, after)
    return diff > threshold  # 有变化则认为成功
```

**简化方案**：MVP阶段，不检测，假设成功。如果用户反馈操作无效，再触发SendInput确认流程。

## 方案2：SendInput (pyautogui)

### 原理

模拟硬件级别的输入，系统无法区分真实键鼠和模拟输入。

### 依赖

```txt
pyautogui>=0.9.54
```

### 代码示例

```python
import pyautogui
import time

class SendInputInjector:

    def click(self, x: int, y: int) -> bool:
        """点击操作"""
        pyautogui.click(x, y)
        return True

    def right_click(self, x: int, y: int) -> bool:
        """右键点击"""
        pyautogui.rightClick(x, y)
        return True

    def long_press(self, x: int, y: int, duration_ms: int) -> bool:
        """长按操作"""
        pyautogui.moveTo(x, y)
        pyautogui.mouseDown()
        time.sleep(duration_ms / 1000)
        pyautogui.mouseUp()
        return True

    def swipe(self, start_x: int, start_y: int, end_x: int, end_y: int) -> bool:
        """滑动操作"""
        duration = 0.3  # 滑动时长
        pyautogui.moveTo(start_x, start_y)
        pyautogui.drag(end_x - start_x, end_y - start_y, duration=duration)
        return True

    def key_press(self, key: str) -> bool:
        """按键操作"""
        pyautogui.hotkey(*self._parse_key(key))
        return True

    def input_text(self, text: str) -> bool:
        """输入文本"""
        pyautogui.typewrite(text)
        return True
```

### 注意事项

- 会移动物理鼠标光标
- 会打断用户当前操作
- **必须弹窗确认**

### 触发条件

## 封装实现

### platform/windows/input.py

```python
from abc import ABC, abstractmethod
from typing import Tuple

class InjectResult:
    success: bool
    error: Optional[str]
    method: str  # "postmessage" | "sendinput"
    need_confirm: bool  # 是否需要用户确认

class WindowsInputInjector:

    def __init__(self, confirm_callback):
        self.confirm_callback = confirm_callback  # Tauri弹窗回调

    def click(self, hwnd: int, x: int, y: int) -> InjectResult:
        """点击操作"""

        # 方案1: PostMessage
        if self._postmessage_click(hwnd, x, y):
            return InjectResult(
                success=True,
                method="postmessage",
                need_confirm=False
            )

        # 方案2: 需要确认
        confirmed = self.confirm_callback()
        if not confirmed:
            return InjectResult(
                success=False,
                error="用户拒绝操作",
                method="",
                need_confirm=True
            )

        # 执行SendInput
        self._sendinput_click(x, y)
        return InjectResult(
            success=True,
            method="sendinput",
            need_confirm=True
        )
```

## 坐标转换

百分比坐标 → 绝对坐标

```python
# utils/coordinate.py

def percent_to_absolute(x_percent: float, y_percent: float,
                        window_rect: Tuple[int, int, int, int]) -> Tuple[int, int]:
    """
    百分比坐标转绝对坐标

    Args:
        x_percent: 横坐标百分比 (0-100)
        y_percent: 纵坐标百分比 (0-100)
        window_rect: 窗口矩形 (left, top, right, bottom)

    Returns:
        (x, y) 绝对坐标
    """
    left, top, right, bottom = window_rect
    width = right - left
    height = bottom - top

    x = left + int(width * x_percent / 100)
    y = top + int(height * y_percent / 100)

    return x, y
```
